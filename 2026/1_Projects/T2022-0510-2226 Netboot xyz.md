- Install ubuntu cloud init server

```bash
wget https://cloud-images.ubuntu.com/jammy/current/jammy-server-cloudimg-amd64-disk-kvm.img
qm create 9000 --memory 2048 --net0 virtio,bridge=vmbr0
qm importdisk 9000 jammy-server-cloudimg-amd64-disk-kvm.img local-lvm
qm set 9000 --scsihw virtio-scsi-pci --scsi0 local-lvm:vm-9000-disk-0
qm set 9000 --ide2 local-lvm:cloudinit
qm set 9000 --boot c --bootdisk scsi0
qm set 9000 --serial0 socket --vga serial0
qm template 9000

qm clone 9000 101 --name ubuntu2
qm set 101 --sshkeys id_rsa.pub
qm set 101 --ipconfig0 ip=10.1.1.20/24,gw=10.1.1.1

qm start 101
qm terminal 101

ssh ubuntu@10.1.1.20

```
